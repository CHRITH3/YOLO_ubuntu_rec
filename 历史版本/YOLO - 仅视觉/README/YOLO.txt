│ │ 数据流设计                                                     │ │
│ │                                                                │ │
│ │ // 核心数据结构                                                │ │
│ │ struct KeyPoint {                                              │ │
│ │     float x, y;           // 2D坐标                            │ │
│ │     float confidence;     // 置信度                            │ │
│ │     cv::Point3f pos3d;   // 3D坐标(利用深度)                   │ │
│ │ };                                                             │ │
│ │                                                                │ │
│ │ struct PoseResult {                                            │ │
│ │     cv::Rect bbox;                    // 人体边界框            │ │
│ │     float box_confidence;             // 检测置信度            │ │
│ │     std::vector<KeyPoint> keypoints;  // 17个关键点            │ │
│ │     int person_id;                    // 人物ID                │ │
│ │ };                                                             │ │
│ │                                                                │ │

│ │ 📐 实现步骤                                                    │ │
│ │                                                                │ │
│ │ 第1步: 准备YOLO模型 (30分钟)                                   │ │
│ │                                                                │ │
│ │ 1. 下载预训练模型                                              │ │
│ │ # YOLOv8n-pose (轻量级，适合实时)                              │ │
│ │ wget https://github.com/ultralytics/assets/releases/download/v │ │
│ │ 8.3.0/yolov8n-pose.pt                                          │ │
│ │ 2. 导出为ONNX格式                                              │ │
│ │ from ultralytics import YOLO                                   │ │
│ │ model = YOLO('yolov8n-pose.pt')                                │ │
│ │ model.export(format='onnx', imgsz=640)                         │ │
│ │ # 输出: yolov8n-pose.onnx                                      │ │
│ │ 3. 验证模型                                                    │ │
│ │   - 输入: [1, 3, 640, 640] (NCHW)                              │ │
│ │   - 输出: [1, 56, 8400] (每个检测: 4bbox + 1conf + 1cls +      │ │
│ │ 17x3keypoints)                                                 │ │
│ │                                                                │ │
│ │ 第2步: 安装ONNX Runtime (15分钟)                               │ │
│ │                                                                │ │
│ │ # 下载预编译库                                                 │ │
│ │ wget https://github.com/microsoft/onnxruntime/releases/downloa │ │
│ │ d/v1.17.0/onnxruntime-linux-x64-1.17.0.tgz                     │ │
│ │ tar -xzf onnxruntime-linux-x64-1.17.0.tgz                      │ │
│ │ # 配置环境变量或CMake链接                                      │ │
│ │                                                                │ │
│ │ 第3步: 创建推理引擎类 (2小时)                                  │ │
│ │                                                                │ │
│ │ // yolo_pose_detector.h                                        │ │
│ │ class YOLOPoseDetector {                                       │ │
│ │ public:                                                        │ │
│ │     YOLOPoseDetector(const std::string& model_path);           │ │
│ │     std::vector<PoseResult> detect(const cv::Mat& image);      │ │
│ │                                                                │ │
│ │ private:                                                       │ │
│ │     Ort::Session* session_;                                    │ │
│ │     void preprocess(const cv::Mat& img, std::vector<float>&    │ │
│ │ input_data);                                                   │ │
│ │     std::vector<PoseResult> postprocess(float* output,         │ │
│ │                                         cv::Size               │ │
│ │ original_size);                                                │ │
│ │     // NMS, 坐标转换, 关键点解析                               │ │
│ │ };                                                             │ │
│ │                                                                │ │
│ │ 关键实现点:                                                    │ │
│ │ - 预处理: BGR→RGB, Resize, Normalize, HWC→CHW                  │ │
│ │ - 推理: ONNX Runtime API调用                                   │ │
│ │ - 后处理:                                                      │ │
│ │   - NMS去重 (IoU阈值0.45)                                      │ │
│ │   - 置信度筛选 (>0.5)                                          │ │
│ │   - 关键点坐标转换回原图尺寸                                   │ │
│ │   - 关键点可见性筛选 (confidence>0.5)                          │ │
│ │                                                                │ │
│ │ 第4步: 集成到SDK (2小时)                                       │ │
│ │                                                                │ │
│ │ 创建 get_pose_with_depth.cpp:                                  │ │
│ │                                                                │ │
│ │ // 核心流程                                                    │ │
│ │ main() {                                                       │ │
│ │     // 1. 初始化SDK                                            │ │
│ │     CIMRSDK* sdk = new CIMRSDK();                              │ │
│ │     sdk->Init(config);                                         │ │
│ │                                                                │ │
│ │     // 2. 初始化YOLO Pose                                      │ │
│ │     YOLOPoseDetector detector("yolov8n-pose.onnx");            │ │
│ │                                                                │ │
│ │     // 3. 获取相机内参(用于3D映射)                             │ │
│ │     auto params = sdk->GetModuleParams();                      │ │
│ │                                                                │ │
│ │     // 4. 注册回调                                             │ │
│ │     sdk->RegistImgCallback([&](double time, cv::Mat left,      │ │
│ │ cv::Mat right) {                                               │ │
│ │         image_queue.push(left);  // 使用左图                   │ │
│ │     });                                                        │ │
│ │                                                                │ │
│ │     sdk->EnableDepthProcessor();                               │ │
│ │     sdk->RegistDepthCallback([&](double time, cv::Mat depth) { │ │
│ │         depth_queue.push(depth);                               │ │
│ │     });                                                        │ │
│ │                                                                │ │
│ │     // 5. 主循环                                               │ │
│ │     while (true) {                                             │ │
│ │         if (!image_queue.empty() && !depth_queue.empty()) {    │ │
│ │             cv::Mat img = image_queue.front();                 │ │
│ │             cv::Mat depth = depth_queue.front();               │ │
│ │                                                                │ │
│ │             // YOLO推理                                        │ │
│ │             auto poses = detector.detect(img);                 │ │
│ │                                                                │ │
│ │             // 3D坐标映射                                      │ │
│ │             for (auto& pose : poses) {                         │ │
│ │                 map_to_3d(pose, depth, params);                │ │
│ │             }                                                  │ │
│ │                                                                │ │
│ │             // 可视化                                          │ │
│ │             draw_poses(img, poses);                            │ │
│ │             cv::imshow("Pose + Depth", img);                   │ │
│ │         }                                                      │ │
│ │     }                                                          │ │
│ │ }                                                              │ │
│ │                                                                │ │
│ │ 第5步: 深度融合与3D映射 (1.5小时)                              │ │
│ │                                                                │ │
│ │ void map_to_3d(PoseResult& pose, const cv::Mat& depth,         │ │
│ │                const CameraParams& params) {                   │ │
│ │     for (auto& kp : pose.keypoints) {                          │ │
│ │         // 获取2D点的深度                                      │ │
│ │         int x = cvRound(kp.x);                                 │ │
│ │         int y = cvRound(kp.y);                                 │ │
│ │         float Z = depth.at<float>(y, x) * 1000;  // 转为毫米   │ │
│ │                                                                │ │
│ │         // 投影到3D                                            │ │
│ │         kp.pos3d.x = (x - params.cx) * Z / params.fx;          │ │
│ │         kp.pos3d.y = (y - params.cy) * Z / params.fy;          │ │
│ │         kp.pos3d.z = Z;                                        │ │
│ │     }                                                          │ │
│ │ }                                                              │ │
│ │                                                                │ │
│ │ 第6步: 可视化增强 (1小时)                                      │ │
│ │                                                                │ │
│ │ void draw_poses(cv::Mat& img, const std::vector<PoseResult>&   │ │
│ │ poses) {                                                       │ │
│ │     // COCO-17关键点连接骨架                                   │ │
│ │     static const std::vector<std::pair<int,int>> skeleton = {  │ │
│ │         {0,1}, {0,2}, {1,3}, {2,4},      // 头部               │ │
│ │         {5,6}, {5,7}, {7,9}, {6,8}, {8,10}, // 上肢            │ │
│ │         {5,11}, {6,12}, {11,12},         // 躯干               │ │
│ │         {11,13}, {13,15}, {12,14}, {14,16} // 下肢             │ │
│ │     };                                                         │ │
│ │                                                                │ │
│ │     for (const auto& pose : poses) {                           │ │
│ │         // 1. 画边界框                                         │ │
│ │         cv::rectangle(img, pose.bbox, cv::Scalar(0,255,0), 2); │ │
│ │                                                                │ │
│ │         // 2. 画关键点                                         │ │
│ │         for (const auto& kp : pose.keypoints) {                │ │
│ │             if (kp.confidence > 0.5) {                         │ │
│ │                 cv::circle(img, cv::Point(kp.x, kp.y),         │ │
│ │                           5, cv::Scalar(0,0,255), -1);         │ │
│ │             }                                                  │ │
│ │         }                                                      │ │
│ │                                                                │ │
│ │         // 3. 画骨架连线                                       │ │
│ │         for (const auto& [i, j] : skeleton) {                  │ │
│ │             if (pose.keypoints[i].confidence > 0.5 &&          │ │
│ │                 pose.keypoints[j].confidence > 0.5) {          │ │
│ │                 cv::line(img,                                  │ │
│ │                         cv::Point(pose.keypoints[i].x,         │ │
│ │ pose.keypoints[i].y),                                          │ │
│ │                         cv::Point(pose.keypoints[j].x,         │ │
│ │ pose.keypoints[j].y),                                          │ │
│ │                         cv::Scalar(255,255,0), 2);             │ │
│ │             }                                                  │ │
│ │         }                                                      │ │
│ │                                                                │ │
│ │         // 4. 显示3D信息                                       │ │
│ │         std::stringstream ss;                                  │ │
│ │         ss << "Person " << pose.person_id                      │ │
│ │            << " Depth: " << pose.keypoints[0].pos3d.z << "mm"; │ │
│ │         cv::putText(img, ss.str(), ...);                       │ │
│ │     }                                                          │ │
│ │ }                                                              │ │

│ │ 🎯 关键技术点                                                  │ │
│ │                                                                │ │
│ │ 1. 17关键点定义 (COCO格式)                                     │ │
│ │                                                                │ │
│ │ 0-鼻子, 1-左眼, 2-右眼, 3-左耳, 4-右耳,                        │ │
│ │ 5-左肩, 6-右肩, 7-左肘, 8-右肘, 9-左腕, 10-右腕,               │ │
│ │ 11-左髋, 12-右髋, 13-左膝, 14-右膝, 15-左踝, 16-右踝           │ │
│ │                                                                │ │
│ │ 2. YOLO输出解析                                                │ │
│ │                                                                │ │
│ │ // 输出shape: [1, 56, 8400]                                    │ │
│ │ // 56维: [4bbox, 1objectness, 1class, 17x3keypoints]           │ │
│ │ for (int i = 0; i < 8400; i++) {                               │ │
│ │     float* ptr = output + i * 56;                              │ │
│ │     float cx = ptr[0], cy = ptr[1], w = ptr[2], h = ptr[3];    │ │
│ │     float obj_conf = ptr[4];                                   │ │
│ │     float cls_conf = ptr[5];  // person类                      │ │
│ │                                                                │ │
│ │     if (obj_conf * cls_conf > 0.5) {                           │ │
│ │         for (int k = 0; k < 17; k++) {                         │ │
│ │             float kp_x = ptr[6 + k*3];                         │ │
│ │             float kp_y = ptr[6 + k*3 + 1];                     │ │
│ │             float kp_conf = ptr[6 + k*3 + 2];                  │ │
│ │             // 保存关键点...                                   │ │
│ │         }                                                      │ │
│ │     }                                                          │ │
│ │ }                                                              │ │
│ │                                                                │ │
│ │ 3. 深度图同步                                                  │ │
│ │                                                                │ │
│ │ // 确保图像和深度时间戳对齐                                    │ │
│ │ if (abs(img_time - depth_time) < 0.02) {  // 20ms容差          │ │
│ │     process_frame(img, depth);                                 │ │
│ │ }                                                              │ │
